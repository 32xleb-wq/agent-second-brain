### Что делаем ПРЯМО СЕЙЧАС (1 шаг)
Начнём с **Aave v3** (самый быстрый выигрыш). Ниже — компактная функция для Apps Script, которая вернёт Health Factor по адресу кошелька.

1) В таблице: «Расширения → Apps Script» → вставь код и **сохрани**.

```javascript
// === AAVE Health Factor (v3) ===
// Вызов из ячейки: =AAVE_HF("RPC_URL", "POOL_ADDRESS", "USER_ADDRESS")
function AAVE_HF(rpcUrl, poolAddress, userAddress) {
  if (!rpcUrl || !poolAddress || !userAddress) return "ARGS";
  // selector getUserAccountData(address) = 0xb6... (см. пояснение ниже)
  var selector = "0xb53e1d7e"; // сигнатура метода (getUserAccountData(address))
  var clean = userAddress.toLowerCase().replace(/^0x/, "");
  var padded = ("0".repeat(64) + clean).slice(-64);
  var data = selector + padded;

  var payload = {
    jsonrpc: "2.0",
    id: 1,
    method: "eth_call",
    params: [{ to: poolAddress, data: "0x" + data.replace(/^0x/, "") }, "latest"]
  };

  var res = UrlFetchApp.fetch(rpcUrl, {
    method: "post",
    contentType: "application/json",
    muteHttpExceptions: true,
    payload: JSON.stringify(payload)
  });
  var out = JSON.parse(res.getContentText());
  if (!out.result) return "ERR";

  // Ответ — 6*32 байт; последний слот = healthFactor (uint256, 1e18)
  var hex = out.result.replace(/^0x/, "");
  var hfHex = hex.slice(64 * 5, 64 * 6);
  var hf = parseInt(hfHex, 16) / 1e18;
  return hf;
}
```

```javascript
// ==== CLEAN IMPORTJSON (stable) ====
// Возвращает: значение по JSON-пути либо "PATH_NULL"/"NOT_JSON"
function IMPORTJSON(url, path) {
  if (!url) return "";
  var resp = UrlFetchApp.fetch(String(url), { muteHttpExceptions: true });
  var text = resp.getContentText();
  var json;
  try { json = JSON.parse(text); }
  catch (e) { return "NOT_JSON"; }

  var parts = String(path || "").split("/").filter(function(s){ return s !== ""; });
  var node = json;

  for (var i = 0; i < parts.length; i++) {
    var key = parts[i];

    if (Array.isArray(node)) {
      var idx = parseInt(key, 10);
      if (isNaN(idx) || idx < 0 || idx >= node.length) return "PATH_NULL";
      node = node[idx];
    } else if (node && typeof node === "object") {
      if (!(key in node) || node[key] == null) return "PATH_NULL";
      node = node[key];
    } else {
      return "PATH_NULL";
    }
  }
  return (typeof node === "object") ? JSON.stringify(node) : node;
}
```
